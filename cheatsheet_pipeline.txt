# Full pipeline from scratch (charge -1, 3 confs by default)
python pipeline.py --mol molecules/mglyval.xyz --charge -1

# Specify number of conformers
python pipeline.py --mol molecules/mglyval.xyz --charge -1 --nconf 5

# Neutral molecule
python pipeline.py --mol molecules/mol_A.xyz --charge 0

# Resume from OPT (CREST already done)
python pipeline.py --mol molecules/mglyval.xyz --charge -1 --start opt

# Resume from NumFreq (auto-detects restart if freq_D*.res.Gradients is present)
python pipeline.py --mol molecules/mglyval.xyz --charge -1 --start numfreq

# Resume from SP
python pipeline.py --mol molecules/mglyval.xyz --charge -1 --start sp

# Resume from TDDFT
python pipeline.py --mol molecules/mglyval.xyz --charge -1 --start tddft --nconf 3

# Skip NumFreq entirely: OPT → SP → TDDFT directly (normal mode only,
# not compatible with --expand-crest). Uses 02_opt/ geometry directly as
# SP input, with no NumFreq thermal corrections in the pKa cycle.
python pipeline.py --mol molecules/mol_A.xyz --charge -1 --skipnumfreq

# Cheap CENSO pre-screening between CREST and OPT (PBE-D4/SV(p) via
# ~/.censo2rc) to discard high-energy conformers before the costly
# wB97X-D3/SVP OPT. Normal mode only. Fail-open: if CENSO fails, the
# unfiltered CREST ensemble is used.
python pipeline.py --mol molecules/mol_A.xyz --charge -1 --use-censo-prescreen

# SP + NRoots 1 mode: minimal TD-DFT SP (NRoots 1) on the best conformers
# by OPT DFT energy (requires 02_opt/ to be complete). Inputs written to
# 06_sp_nroot/<conf_id>/sp_nroot.inp. Incompatible with --expand-crest.
python pipeline.py --mol molecules/mglyval.xyz --charge -1 --nroot

# Same, but over the top-3 conformers instead of the default top-5
python pipeline.py --mol molecules/mglyval.xyz --charge -1 --nroot --nroot-top 3

# Simple mode: declare total cores available
# (OPT/SP/TDDFT use all of them; NumFreq uses min(--cores, 6))
python pipeline.py --mol molecules/mglyval.xyz --charge -1 --cores 8

# Run two pipelines in parallel without competing for cores (7+7 = 14 cores)
python pipeline.py --mol molecules/mol_A.xyz --charge -1 --cores 7 &
python pipeline.py --mol molecules/mol_B.xyz --charge -1 --cores 7 &

# Fine-grained control per stage
python pipeline.py --mol molecules/mglyval.xyz --charge -1 \
    --nprocs 4 --maxcore 3500 \
    --nprocs-numfreq 6 --maxcore-numfreq 2300

# Only change memory (useful under RAM pressure)
python pipeline.py --mol molecules/mglyval.xyz --charge -1 \
    --maxcore 2500 --maxcore-numfreq 1800

# Expand with defaults (skip=3, window=17, threshold=3.0 kcal/mol, 3 parallel OPTs)
python pipeline.py --mol molecules/mol_B.xyz --charge -1 --expand-crest

# Custom expand: skip the first 5 confs, explore 20 more
python pipeline.py --mol molecules/mol_B.xyz --charge -1 --expand-crest \
    --crest-skip 5 --crest-window 20

# Change the ΔE filter threshold
python pipeline.py --mol molecules/mol_B.xyz --charge -1 --expand-crest \
    --dft-threshold 2.0

# Control expand parallelism: 3 OPTs × 4 cores = 12 total cores
python pipeline.py --mol molecules/mol_B.xyz --charge -1 --expand-crest \
    --cores 12 --parallel 3

# Resume expand from NumFreq (expand OPTs already done)
python pipeline.py --mol molecules/mol_B.xyz --charge -1 --expand-crest \
    --start numfreq

# Full expand with all parameters explicit
python pipeline.py --mol molecules/mol_B.xyz --charge -1 --expand-crest \
    --crest-skip 3 --crest-window 17 \
    --dft-threshold 3.0 \
    --parallel 3 \
    --cores 12

# See status of all runs
python pipeline.py --status
