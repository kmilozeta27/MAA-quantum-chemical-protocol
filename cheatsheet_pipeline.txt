# ══════════════════════════════════════════════════════════════
#  pipeline.py — CHEATSHEET
#  Flujo: CREST → CENSO (prescreen, automático) → OPT (nconf auto)
#         → NumFreq → SP → TDDFT → ADC2
#
#  Notas clave:
#    • CENSO pre-screening SIEMPRE corre entre CREST y OPT (fail-open).
#    • nconf se determina automáticamente del ensemble post-CENSO (mín 3, máx 20).
#    • --start crest y --start opt son equivalentes al inicio.
#      Usa --start opt/numfreq/sp/tddft/adc2 solo para retomar con CREST+CENSO ya hechos.
# ══════════════════════════════════════════════════════════════

# ─────────────────────────────────────────────
#  MODO NORMAL
# ─────────────────────────────────────────────

# Full pipeline desde cero (charge -1)
python pipeline.py --mol molecules/mglyval.xyz --charge -1

# Molécula neutra
python pipeline.py --mol molecules/mol_A.xyz --charge 0

# Retomar desde OPT (CREST+CENSO ya hechos)
python pipeline.py --mol molecules/mglyval.xyz --charge -1 --start opt

# Retomar desde NumFreq
python pipeline.py --mol molecules/mglyval.xyz --charge -1 --start numfreq

# Retomar desde SP
python pipeline.py --mol molecules/mglyval.xyz --charge -1 --start sp

# Retomar desde TDDFT
python pipeline.py --mol molecules/mglyval.xyz --charge -1 --start tddft

# Retomar desde ADC2
python pipeline.py --mol molecules/mglyval.xyz --charge -1 --start adc2

# Skip NumFreq: OPT → SP → TDDFT → ADC2 directo (modo normal solo;
# incompatible con --expand-crest). SP usa geometría de 02_opt/ directamente,
# sin correcciones térmicas de NumFreq en el ciclo pKa.
python pipeline.py --mol molecules/mol_A.xyz --charge -1 --skipnumfreq

# SP + NRoots 1: TD-DFT mínimo sobre los top-N confórmeros por E_OPT.
# Resultados en 06_sp_nroot/. Requiere 02_opt/ completo. Incompatible con --expand-crest.
python pipeline.py --mol molecules/mglyval.xyz --charge -1 --nroot
python pipeline.py --mol molecules/mglyval.xyz --charge -1 --nroot --nroot-top 3  # top-3 en vez de top-5

# Ver estado de todos los runs
python pipeline.py --status

# ─────────────────────────────────────────────
#  RECURSOS DE CÓMPUTO
# ─────────────────────────────────────────────

# El pipeline autodetecta CPU y RAM; sin flags extra, usa la topología real.
# --cores N: declara los cores totales disponibles (recomendado).
#   El script distribuye automáticamente por etapa según ResourceProfile.
python pipeline.py --mol molecules/mglyval.xyz --charge -1 --cores 8

# Dos pipelines en paralelo sin competir por cores (7+7 = 14 cores)
python pipeline.py --mol molecules/mol_A.xyz --charge -1 --cores 7 &
python pipeline.py --mol molecules/mol_B.xyz --charge -1 --cores 7 &

# Control fino por etapa (avanzado; sobreescribe --cores)
#   --nprocs          cores/job para OPT / SP / TDDFT
#   --maxcore         MB/core   para OPT / SP / TDDFT
#   --nprocs-numfreq  cores para NumFreq
#   --maxcore-numfreq MB/core para NumFreq
python pipeline.py --mol molecules/mglyval.xyz --charge -1 \
    --nprocs 4 --maxcore 3500 \
    --nprocs-numfreq 6 --maxcore-numfreq 2300

# Solo ajustar memoria (bajo presión de RAM)
python pipeline.py --mol molecules/mglyval.xyz --charge -1 \
    --maxcore 2500 --maxcore-numfreq 1800

# ─────────────────────────────────────────────
#  MODO EXPANSIÓN CONFORMACIONAL (--expand-crest)
# ─────────────────────────────────────────────
#  Diseñado para casos con inversión energética GFN2→DFT (ej. estado B).
#  Extrae confs CREST[skip:skip+window], los optimiza en lotes paralelos,
#  filtra por ΔE(DFT) < threshold y continúa: NUMFREQ → SP → TDDFT → ADC2.
#  Requiere que CREST haya completado (01_crest/ debe existir).
#  Incompatible con --skipnumfreq y --nroot.

# Expansión con defaults (skip=3, window=17, threshold=3.0 kcal/mol)
# Autodetecta recursos; ResourceProfile reparte cores entre --parallel jobs.
python pipeline.py --mol molecules/mol_B.xyz --charge -1 --expand-crest

# Declarar cores disponibles (recomendado con --expand-crest)
python pipeline.py --mol molecules/mol_B.xyz --charge -1 --expand-crest --cores 12

# Control de parallelismo: 2 OPTs simultáneos × 6 cores c/u (= 12 cores)
python pipeline.py --mol molecules/mol_B.xyz --charge -1 --expand-crest \
    --cores 12 --parallel 2

# Ventana personalizada: saltarse los primeros 5, explorar 20 más
python pipeline.py --mol molecules/mol_B.xyz --charge -1 --expand-crest \
    --crest-skip 5 --crest-window 20

# Cambiar umbral ΔE del filtro post-OPT
python pipeline.py --mol molecules/mol_B.xyz --charge -1 --expand-crest \
    --dft-threshold 2.0

# Todos los parámetros explícitos
python pipeline.py --mol molecules/mol_B.xyz --charge -1 --expand-crest \
    --crest-skip 3 --crest-window 17 \
    --dft-threshold 3.0 \
    --parallel 3 \
    --cores 12

# Retomar expand desde NumFreq (expand-OPT ya completado)
# Requiere expand_filter_result.json de la corrida OPT previa.
python pipeline.py --mol molecules/mol_B.xyz --charge -1 --expand-crest \
    --start numfreq

# ─────────────────────────────────────────────
#  MONITOREO
# ─────────────────────────────────────────────

# Seguir el log en vivo de un run
tail -f runs/mglyval/pipeline.log

# Monitorear uso de cores
htop

# Ver todos los procesos ORCA activos
ps aux | grep orca

# Contar confórmeros generados por CREST
python3 -c "from ase.io import read; print(len(read('runs/mglyval/01_crest/crest_conformers.xyz', index=':')))"

# ─────────────────────────────────────────────
#  EJEMPLOS RÁPIDOS
# ─────────────────────────────────────────────

# M-Gly:Val, charge -1, flujo completo
python pipeline.py --mol molecules/mglyval.xyz --charge -1

# Dos moléculas en paralelo, 7 cores cada una
nohup python pipeline.py --mol molecules/mglyval.xyz --charge -1 --cores 7 \
    > mglyval.log 2>&1 &
nohup python pipeline.py --mol molecules/mol_B.xyz --charge -1 --cores 7 \
    > mol_B.log 2>&1 &

# Explorar inversión GFN2→DFT en estado B
python pipeline.py --mol molecules/mol_B.xyz --charge -1 --expand-crest \
    --crest-skip 3 --crest-window 17 --dft-threshold 3.0 \
    --parallel 3 --cores 12
